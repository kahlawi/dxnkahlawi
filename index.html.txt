<script>
const products = [
  {id:1,name:"قهوة لينغزي",category:"coffee",price:25,emoji:"☕",desc:"مثال تجريبي لمنتج من فئة القهوة."},
  {id:2,name:"قهوة بلاك كوفي",category:"coffee",price:18,emoji:"☕",desc:"مثال تجريبي. بدّل الاسم والسعر حسب المنتج الحقيقي."},
  {id:3,name:"شاي عشبي",category:"tea",price:20,emoji:"🍵",desc:"مثال تجريبي لمنتج من فئة الشاي."},
  {id:4,name:"مشروب الكاكاو",category:"coffee",price:22,emoji:"🥤",desc:"مثال تجريبي لمشروب."},
  {id:5,name:"منتج عناية بالبشرة",category:"care",price:30,emoji:"🧴",desc:"مثال تجريبي لمنتج العناية الشخصية."},
  {id:6,name:"صابون للعناية",category:"care",price:12,emoji:"🧼",desc:"مثال تجريبي. تأكد من الاسم والسعر قبل النشر."}
];
const cart = {};
const money = n => n.toFixed(2) + " د.ت";
const list = document.getElementById("productsList");

function renderProducts(){
  const q = document.getElementById("search").value.trim().toLowerCase();
  const cat = document.getElementById("category").value;
  const filtered = products.filter(p =>
    p.name.toLowerCase().includes(q) &&
    (cat === "all" || p.category === cat)
  );
  list.innerHTML = "";
  if(!filtered.length){
    list.textContent = "ما لقيناش منتجات مطابقة للبحث.";
    return;
  }
  filtered.forEach(p => {
    const card = document.createElement("article");
    card.className = "card";
    const image = document.createElement("div");
    image.className = "product-img";
    image.textContent = p.emoji;
    const content = document.createElement("div");
    content.className = "card-content";
    const title = document.createElement("h3");
    title.textContent = p.name;
    const desc = document.createElement("p");
    desc.textContent = p.desc;
    const price = document.createElement("div");
    price.className = "price";
    price.textContent = money(p.price);
    const btn = document.createElement("button");
    btn.className = "green";
    btn.textContent = "أضف للسلة";
    btn.onclick = () => { cart[p.id] = (cart[p.id] || 0) + 1; renderCart(); };
    content.append(title,desc,price,btn);
    card.append(image,content);
    list.append(card);
  });
}

function renderCart(){
  const container = document.getElementById("cartItems");
  container.innerHTML = "";
  const ids = Object.keys(cart).filter(id => cart[id] > 0);
  let total = 0;
  if(!ids.length){
    container.textContent = "السلة فارغة توّا.";
  }
  ids.forEach(id => {
    const p = products.find(x => x.id === Number(id));
    const qty = cart[id];
    total += p.price * qty;
    const row = document.createElement("div");
    row.className = "cart-row";
    const name = document.createElement("span");
    name.textContent = p.name + " — " + money(p.price);
    const controls = document.createElement("div");
    controls.className = "qty";
    const minus = document.createElement("button");
    minus.textContent = "−";
    minus.setAttribute("aria-label","نقص الكمية");
    minus.onclick = () => { cart[id]--; renderCart(); };
    const number = document.createElement("span");
    number.textContent = qty;
    const plus = document.createElement("button");
    plus.textContent = "+";
    plus.setAttribute("aria-label","زيد الكمية");
    plus.onclick = () => { cart[id]++; renderCart(); };
    const subtotal = document.createElement("strong");
    subtotal.textContent = money(p.price * qty);
    controls.append(minus,number,plus);
    row.append(name,controls,subtotal);
    container.append(row);
  });
  document.getElementById("total").textContent = "المجموع: " + money(total);
}

function sendOrder(){
  const ids = Object.keys(cart).filter(id => cart[id] > 0);
  if(!ids.length){ alert("زيد منتج واحد على الأقل للسلة."); return; }
  const name = document.getElementById("customerName").value.trim();
  const phone = document.getElementById("customerPhone").value.trim();
  const city = document.getElementById("customerCity").value.trim();
  const address = document.getElementById("customerAddress").value.trim();
  if(!name || !phone || !city){
    alert("عمّر الاسم ورقم الهاتف والولاية/المدينة من فضلك.");
    return;
  }
  let total = 0;
  const lines = ids.map(id => {
    const p = products.find(x => x.id === Number(id));
    total += p.price * cart[id];
    return "• " + p.name + " × " + cart[id] + " = " + money(p.price * cart[id]);
  });
  const message = [
    "السلام عليكم، نحب نأكد الطلب من DXNkahlawi:",
    "",
    ...lines,
    "",
    "المجموع التقريبي: " + money(total),
    "الاسم: " + name,
    "الهاتف: " + phone,
    "المدينة/الولاية: " + city,
    address ? "العنوان: " + address : "",
    "",
    "الرجاء تأكيد توفّر المنتجات ومصاريف التوصيل."
  ].filter(Boolean).join("\n");
  window.open("https://wa.me/21655286082?text=" + encodeURIComponent(message),"_blank","noopener");
}

document.getElementById("search").addEventListener("input",renderProducts);
document.getElementById("category").addEventListener("change",renderProducts);
document.getElementById("year").textContent = new Date().getFullYear();
renderProducts();
renderCart();
